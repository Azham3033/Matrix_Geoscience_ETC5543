# Load parcels for all municipality shapefiles

# Each dir must contain a shapefile named "<folder>.shp"
shp_paths <- file.path(municipality_dirs, paste0(basename(municipality_dirs), ".shp"))
shp_paths <- shp_paths[file.exists(shp_paths)]

# Name each path by its municipality (no dot pronoun with base pipe)
names(shp_paths) <- basename(municipality_dirs)[file.exists(file.path(municipality_dirs, paste0(basename(municipality_dirs), ".shp")))]

# Read all shapefiles into a named list (use \(p) for base pipe)
parcels_list <- shp_paths |>
  map(\(p) st_read(p, quiet = TRUE))

# Combine & filter parcels data
parcels_all <- bind_rows(parcels_list, .id = "Municipality") |>
  filter(PID != "0", TENURE_TY != "Crown Land", MEAS_AREA >= 100000) |>
  mutate(
    Municipality = factor(Municipality, levels = sort(unique(Municipality))),
    Municipality = as.integer(Municipality)
  ) |>
  select(-any_of(c("VOLUME","FEAT_NAME","STRATA_LEV","COMP_AREA","MEAS_AREA",
                   "UFI","FMP","CREATED_ON","SHAPE_AREA","SHAPE_LEN"))) |>
  distinct(PID, .keep_all = TRUE)

# Load LIST address points and build FULL_ADDRESS
address_points <- st_read(addr_path, quiet = TRUE) |>
  select(PID, ST_NO_FROM, STREET, ST_TYPE, LOCALITY, STATE, POSTCODE, geometry) |>
  mutate(FULL_ADDRESS = str_squish(str_c(ST_NO_FROM, STREET, ST_TYPE, LOCALITY, STATE, POSTCODE, sep = " "))) |>
  select(PID, FULL_ADDRESS, geometry)

# Join parcels with address points on PID
joined_data <- parcels_all |>
  inner_join(st_drop_geometry(address_points), by = "PID")

# Load ESM raster & VAT
r   <- rast(r_path)
vat <- read.dbf(vat_path, as.is = TRUE)

val_col <- grep("^value$", names(vat), ignore.case = TRUE, value = TRUE)
if (length(val_col) != 1L) stop("Could not identify VAT 'Value' column.")

if (length(lab_col) == 0L) lab_col <- val_col

# Reproject parcel to raster CRS
parcel_r <- st_transform(parcel, st_crs(r))

# Exact extract over parcel with majority rule (>= 50% coverage)
cell_df  <- exact_extract(r, parcel_r, fun = NULL, progress = FALSE)[[1]]
val_name <- setdiff(names(cell_df), "coverage_fraction")
if (length(val_name) != 1L) stop("Unexpected raster columns from exact_extract().")
cell_df  <- rename(cell_df, value = !!val_name)

vat_lu_dedup <- distinct(vat_lu, code, label)

majority_tab <- cell_df |>
  ungroup() |>                                        # ensure no residual grouping
  filter(!is.na(value), coverage_fraction >= 0.5) |>
  count(value, name = "COUNT") |>                     # count by raster code
  left_join(vat_lu_dedup, by = c("value" = "code")) |>
  group_by(CLASS_LABEL = label) |>                    # collapse to human-readable class
  summarise(COUNT = sum(COUNT), .groups = "drop") |>
  mutate(
    PID          = parcel$PID,
    FULL_ADDRESS = parcel$FULL_ADDRESS,
    CROP         = crop_name
  ) |>
  select(PID, FULL_ADDRESS, COUNT, CLASS_LABEL, CROP) |>
  arrange(desc(COUNT))

# Show total pixels (majority-rule)
cat(glue::glue("Total pixels (majority-rule): {sum(majority_tab$COUNT)}\n\n"))

# Build a compact list of "matches" to email (top per class, example rule)
matches <- majority_tab %>%
  filter(grepl("well|suit", CLASS_LABEL, ignore.case = TRUE)) %>%
  group_by(PID, FULL_ADDRESS, CROP) %>%
  summarise(TOTAL_PIX = sum(COUNT), .groups = "drop") %>%
  arrange(desc(TOTAL_PIX)) %>%
  slice_head(n = 10)

send_match_email <- function(matches, to, crop_name = unique(matches$CROP),
                             shiny_base = "https://your-shiny.app/parcel",
                             dry_run = TRUE) {
  if (nrow(matches) == 0) { message("No matches → no email."); return(invisible(NULL)) }

  table_md <- paste0(
    "| Address | PID | Pixels | Link |\n|:--|--:|--:|:--|\n",
    paste0(
      "| ", matches$FULL_ADDRESS, " | ", matches$PID, " | ", matches$TOTAL_PIX,
      " | [Open](", shiny_base, "?pid=", matches$PID, "&crop=", crop_name, ") |",
      collapse = "\n"
    )
  )
  msg <- compose_email(
    body = md(glue("
**New {crop_name} matches ({nrow(matches)})**

{table_md}

Generated {format(Sys.time(), '%Y-%m-%d %H:%M %Z')}
"))
  )

  if (dry_run) { print(msg); return(invisible(msg)) }

  smtp <- smtp_server(
    host = Sys.getenv("EMAIL_SMTP_HOST"),
    port = as.integer(Sys.getenv("EMAIL_SMTP_PORT")),
    username = Sys.getenv("EMAIL_USER"),
    password = Sys.getenv("EMAIL_PASS"),
    use_ssl = TRUE
  )
  smtp_send(
    email = msg,
    from  = Sys.getenv("EMAIL_FROM"),
    to    = to,
    subject = sprintf("New %s matches (%d)", crop_name[1], nrow(matches)),
    credentials = smtp
  )
}

escape_html <- function(x) {
  x <- gsub("&","&amp;",x,fixed=TRUE); x <- gsub("<","&lt;",x,fixed=TRUE); gsub(">","&gt;",x,fixed=TRUE)
}

if (!nzchar(tg_chat))  stop("TG_CHAT not set. Use your real chat/channel ID.")
