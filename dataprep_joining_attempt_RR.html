library(exactextractr)  

# Load raster + VAT 
r <- rast("ESM_Tiff/ESM_Tiff/Barley.tif")

vat <- read.dbf("ESM_Tiff/ESM_Tiff/Barley.tif.vat.dbf", as.is = TRUE)

# Bind raster attribute table (RAT) so codes map to labels
val_col <- grep("^value$", names(vat), ignore.case = TRUE, value = TRUE)
if (length(val_col) != 1L) stop("Could not identify VAT 'Value' column in DBF.")

lab_col <- grep("(suitab|class|label|cat)", names(vat), ignore.case = TRUE, value = TRUE)
if (length(lab_col) == 0L) {
  warning("No label-like column found in VAT; labels will equal the numeric code.")
  lab_col <- val_col
}

# Try to find RGB columns for colors
r_col <- grep("^r(ed)?$",   names(vat), ignore.case = TRUE, value = TRUE)
g_col <- grep("^g(reen)?$", names(vat), ignore.case = TRUE, value = TRUE)
b_col <- grep("^b(lue)?$",  names(vat), ignore.case = TRUE, value = TRUE)
has_rgb <- length(r_col) == 1L && length(g_col) == 1L && length(b_col) == 1L

# Build lookup and bind as RAT
rat <- data.frame(
  ID       = vat[[val_col]],
  Category = as.character(vat[[lab_col]]),
  stringsAsFactors = FALSE
)
levels(r) <- rat  

# Also keep a tidy lookup for joining later
vat_lu <- tibble(
  code  = vat[[val_col]],
  label = as.character(vat[[lab_col]])
)
if (has_rgb) {
  to_hex <- function(r,g,b) sprintf("#%02X%02X%02X", pmin(255, as.integer(r)),
                                    pmin(255, as.integer(g)), pmin(255, as.integer(b)))
  vat_lu$color <- to_hex(vat[[r_col]], vat[[g_col]], vat[[b_col]])
} else {
  vat_lu$color <- NA_character_
}

# Select the address polygon from `joined_data` 
target_addr <- "401 BRESNEHANS ROAD LITTLE SWANPORT TAS 7190"

parcel <- joined_data %>%
  mutate(FULL_ADDRESS_UP = str_squish(str_to_upper(FULL_ADDRESS))) %>%
  filter(FULL_ADDRESS_UP == str_squish(str_to_upper(target_addr))) %>%
  slice(1)  # take first if has multiple PIDs

if (nrow(parcel) == 0L) stop("Address not found in joined_data$FULL_ADDRESS.")

# Re-project polygon to raster CRS
parcel_r <- st_transform(parcel, st_crs(r))

# Extract cell-level values + coverage fractions over the polygon 
#   <layer_name> (the raster code) and coverage_fraction
cell_df <- exact_extract(r, parcel_r, fun = NULL, progress = FALSE)[[1]]

# Standardize the value column name to 'value'
val_name <- setdiff(names(cell_df), c("coverage_fraction"))
if (length(val_name) != 1L) stop("Unexpected columns from exact_extract().")
cell_df <- cell_df %>% rename(value = !!val_name)

# Majority rule (>= 50% covered) + tabulate by class 
majority_tab <- cell_df %>%
  filter(!is.na(value), coverage_fraction >= 0.5) %>%  # 50% threshold
  group_by(value) %>%
  summarise(pixels = n(), .groups = "drop") %>%
  left_join(vat_lu, by = c("value" = "code")) %>%
  arrange(desc(pixels)) %>%
  select(class_code = value, class_label = label, color, pixels)

# Identify VAT keys (already loaded: `vat`, `cell_df`)
val_col <- grep("^value$", names(vat), ignore.case = TRUE, value = TRUE)
if (length(val_col) != 1L) stop("Could not identify VAT 'Value' (code) column.")

# pick a human-readable label column (e.g., Suitabilit, CLASS, Label, Category)
lab_col <- grep("(suitab|class|label|cat)", names(vat), ignore.case = TRUE, value = TRUE)[1]
if (is.na(lab_col)) {
  warning("No label-like column found in VAT; falling back to numeric Value as label.")
  lab_col <- val_col
}

# Ensure the extraction's raster column is named 'value'
cell_df <- cell_df %>% rename(value = !!val_name)

# Join to VAT to map internal codes -> logical class 
cell_df2 <- cell_df %>%
  dplyr::left_join(vat %>% dplyr::select(all_of(c(val_col, lab_col))),
                   by = c("value" = val_col))

# Majority rule (>= 50% covered) & collapse by class label
majority_tab <- cell_df2 %>%
  dplyr::filter(!is.na(.data[[lab_col]]), coverage_fraction >= 0.5) %>%
  dplyr::group_by(class_label = .data[[lab_col]]) %>%
  dplyr::summarise(pixels = dplyr::n(), .groups = "drop") %>%
  dplyr::arrange(dplyr::desc(pixels))

# print
print(majority_tab, n = nrow(majority_tab), width = Inf)
